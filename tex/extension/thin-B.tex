\begin{definition}
    A correspondence $V$ is \emph{thin-B} if it is the smallest correspondence that contains a single-valued function $v$ satisfying B and
    \begin{align*}
        V(\mu) &:= \text{Conv} \left( \{ w \in \mathbb{R}: \exists (\mu_n)_{n \in \mathbb{N}} : \mu_n \rightarrow \mu, v(\mu_n) \rightarrow w \} \right).
    \end{align*}
\end{definition}

\begin{lemma}
    A thin-B correspondence $V$ satisfies A4, that is, it is upper hemicontinuous with non-empty compact interval values.
\end{lemma}
