\section*{Thin-B correspondences}

We recall the standing regularity assumption on the payoff correspondence from
the main text.
\begin{assumption}[A4]\label{A4:V-uhc}
    $V$ is upper hemicontinuous with non-empty compact interval values.
\end{assumption}

Fix a finite type space $\Theta$ and let $\Delta\Theta$ denote the probability
simplex, a compact convex subset of $\mathbb{R}^\Theta$. For $\theta\in\Theta$,
write $\delta_\theta\in\Delta\Theta$ for the Dirac belief at $\theta$. Recall
that a single-valued payoff $v:\Delta\Theta\to\mathbb{R}$ satisfies
\emph{betweenness} (B) if, for all $\mu,\mu'\in\Delta\Theta$ and $\lambda\in(0,1)$,
\[
    \min\{v(\mu),v(\mu')\}
    \leq
    v(\lambda\mu+(1-\lambda)\mu')
    \leq
    \max\{v(\mu),v(\mu')\}.
\]
We do \emph{not} assume $v$ is continuous. For $\mu\in\Delta\Theta$, let
\[
    L_v(\mu)
    :=
    \bigl\{
        w\in\mathbb{R}
        \sep
        \exists\,(\mu_n)_{n\in\mathbb{N}}\subseteq\Delta\Theta,\
        \mu_n\to\mu\ \text{ and }\ v(\mu_n)\to w
    \bigr\}
\]
be the set of subsequential limit values of $v$ along sequences in $\Delta\Theta$
approaching $\mu$. The constant sequence $\mu_n\equiv\mu$ shows $v(\mu)\in L_v(\mu)$.

\begin{definition}\label{dfn:thinB}
    Let $v:\Delta\Theta\to\mathbb{R}$ satisfy B. The \emph{thin-B correspondence
    generated by $v$} is the correspondence $V:\Delta\Theta\rightrightarrows\mathbb{R}$
    defined by
    \[
        V(\mu):=\operatorname{Conv}L_v(\mu).
    \]
    A correspondence $V$ is \emph{thin-B} if $V$ is generated by some
    $v$ satisfying B.
\end{definition}

\begin{lemma}\label{lem:thinB-A4}\ \arileanpending
    Every thin-B correspondence $V$ satisfies \cref{A4:V-uhc}: it is upper
    hemicontinuous with non-empty compact interval values.
\end{lemma}

\begin{proof}
    Let $v:\Delta\Theta\to\mathbb{R}$ satisfy B and let $V$ be the thin-B
    correspondence it generates. As noted above, $v(\mu)\in L_v(\mu)$ for every
    $\mu$, so each $L_v(\mu)$ is non-empty.

    \emph{Step 1: $v$ is bounded.} We show by induction on $k$ that if
    $\mu=\sum_{i=1}^{k}\lambda_i\mu_i$ with $\mu_i\in\Delta\Theta$, $\lambda_i>0$,
    and $\sum_i\lambda_i=1$, then
    \[
        \min_{1\leq i\leq k}v(\mu_i)\ \leq\ v(\mu)\ \leq\ \max_{1\leq i\leq k}v(\mu_i).
        \tag{$\ast$}
    \]
    The case $k=1$ is trivial and $k=2$ is B (the endpoint cases
    $\lambda\in\{0,1\}$ being trivial). For $k\geq3$, write
    $\mu=\lambda_1\mu_1+(1-\lambda_1)\mu'$ with
    $\mu':=\sum_{i\geq2}\frac{\lambda_i}{1-\lambda_1}\mu_i\in\Delta\Theta$; the
    inductive hypothesis bounds $v(\mu')$ between $\min_{i\geq2}v(\mu_i)$ and
    $\max_{i\geq2}v(\mu_i)$, and B applied to $\mu_1,\mu'$ bounds $v(\mu)$ between
    $\min_i v(\mu_i)$ and $\max_i v(\mu_i)$, giving $(\ast)$. Applying $(\ast)$
    with the $\mu_i$ ranging over the vertices $\{\delta_\theta:\mu(\theta)>0\}$
    of $\mu$ yields
    \[
        a:=\min_{\theta\in\Theta}v(\delta_\theta)
        \ \leq\ v(\mu)\ \leq\
        \max_{\theta\in\Theta}v(\delta_\theta)=:b
        \qquad\text{for all }\mu\in\Delta\Theta.
    \]
    Hence $v(\Delta\Theta)\subseteq[a,b]$, and so $L_v(\mu)\subseteq[a,b]$ for
    every $\mu$.

    \emph{Step 2: values are non-empty compact intervals.} Fix $\mu$. We claim
    $L_v(\mu)$ is closed. Let $w_k\in L_v(\mu)$ with $w_k\to w$. By the
    definition of $L_v(\mu)$, for each $k$ choose $\nu_k\in\Delta\Theta$ with
    $\lVert\nu_k-\mu\rVert<1/k$ and $\lvert v(\nu_k)-w_k\rvert<1/k$. Then
    $\nu_k\to\mu$ and $v(\nu_k)\to w$, so $w\in L_v(\mu)$. Thus $L_v(\mu)$ is
    closed and, being contained in $[a,b]$, non-empty and compact. Put
    \[
        \underline v(\mu):=\min L_v(\mu),
        \qquad
        \overline v(\mu):=\max L_v(\mu).
    \]
    The convex hull of a non-empty compact subset of $\mathbb{R}$ is the closed
    interval between its minimum and maximum, so
    \[
        V(\mu)=\operatorname{Conv}L_v(\mu)=[\underline v(\mu),\overline v(\mu)],
        \tag{$\dagger$}
    \]
    a non-empty compact interval; in particular $V(\mu)$ is order-connected.

    \emph{Step 3: envelope semicontinuity.} We show $\overline v$ is upper
    semicontinuous and $\underline v$ is lower semicontinuous on $\Delta\Theta$;
    the two arguments are symmetric, so we treat $\overline v$. Let
    $\mu_k\to\mu$ in $\Delta\Theta$ and set $\ell:=\limsup_k\overline v(\mu_k)$;
    we must show $\ell\leq\overline v(\mu)$. Passing to a subsequence, assume
    $\overline v(\mu_k)\to\ell$. Since $\overline v(\mu_k)\in L_v(\mu_k)$, for each
    $k$ choose $\nu_k\in\Delta\Theta$ with $\lVert\nu_k-\mu_k\rVert<1/k$ and
    $\lvert v(\nu_k)-\overline v(\mu_k)\rvert<1/k$. Then $\nu_k\to\mu$ and
    $v(\nu_k)\to\ell$, so $\ell\in L_v(\mu)$ and hence
    $\ell\leq\max L_v(\mu)=\overline v(\mu)$.

    \emph{Step 4: upper hemicontinuity.} Fix $\mu\in\Delta\Theta$ and an open set
    $U\subseteq\mathbb{R}$ with $V(\mu)\subseteq U$. By $(\dagger)$ the interval
    $[\underline v(\mu),\overline v(\mu)]$ is compact and contained in the open set
    $U$, so there is $\varepsilon>0$ with
    $(\underline v(\mu)-\varepsilon,\ \overline v(\mu)+\varepsilon)\subseteq U$. By
    Step 3, the set
    \[
        W:=\bigl\{\mu'\in\Delta\Theta\sep
            \overline v(\mu')<\overline v(\mu)+\varepsilon
            \ \text{ and }\
            \underline v(\mu')>\underline v(\mu)-\varepsilon\bigr\}
    \]
    is a neighbourhood of $\mu$ in $\Delta\Theta$, being the intersection of the
    superlevel neighbourhood given by upper semicontinuity of $\overline v$ and
    the sublevel neighbourhood given by lower semicontinuity of $\underline v$.
    For every $\mu'\in W$, by $(\dagger)$,
    \[
        V(\mu')=[\underline v(\mu'),\overline v(\mu')]
        \subseteq(\underline v(\mu)-\varepsilon,\ \overline v(\mu)+\varepsilon)
        \subseteq U.
    \]
    Hence $V$ is upper hemicontinuous on $\Delta\Theta$.

    Steps 2 and 4 give, respectively, non-empty compact interval values and
    upper hemicontinuity, so $V$ satisfies \cref{A4:V-uhc}.
\end{proof}
